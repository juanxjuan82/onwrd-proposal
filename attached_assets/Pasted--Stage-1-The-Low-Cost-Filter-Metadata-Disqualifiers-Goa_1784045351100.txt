class RFPProcessor:
    def __init__(self, rfp_data: Dict[str, Any]):
        self.rfp = rfp_data
        self.title = rfp_data.get("title", "").lower()
        self.description = rfp_data.get("description", "").lower()
        self.location = rfp_data.get("location", "").lower()
        self.raw_text = f"{self.title} {self.description}".lower()

    def passes_stage1_filter(self) -> Tuple[bool, str]:
        \"\"\"
        Stage 1: Low-Cost Metadata Filter
        Runs rapid regex/string matching to auto-approve local/regional RFPs,

            r"registered vendor in the state of"
        ]

                return False, f"Disqualified: Matched hard exclusion criteria ('{pattern}')"

    def calculate_stage2_score(self) -> Dict[str, Any]:

        Stage 2: High-Fidelity Scoring Matrix
        Calculates a strategic score out of 100 based on ONWRD's positioning.
        \"\"\"
        geo_score = 0
        capability_score = 0
        industry_score = 0
        scale_score = 0

        if "bahamas" in self.location or "nassau" in self.location:
            geo_score = 35
        elif any(country in self.location for country in CARIBBEAN_COUNTRIES):
            geo_score = 28
        else:

            geo_score = 15

        matched_caps = [cap for cap in CORE_CAPABILITIES if cap in self.raw_text]
        cap_ratio = len(matched_caps) / 3 if len(matched_caps) <= 3 else 1.0
        capability_score = int(cap_ratio * 30)

        # --- 3. Industry Vertical Alignment (Max 20 Points) ---
        matched_industries = [ind for ind in TARGET_INDUSTRIES if ind in self.raw_text]
        if matched_industries:
            industry_score = 20
        else:
            industry_score = 5  # Generic baseline if capabilities match but sector varies

        # --- 4. Scale & Feasibility Indicators (Max 15 Points) ---
        # Look for indicators of robust documentation, structured timelines, or clear milestones
        scale_indicators = ["timeline", "milestones", "proposal template", "deliverables", "budget"]
        matched_indicators = sum(1 for ind in scale_indicators if ind in self.raw_text)
        scale_score = int((matched_indicators / len(scale_indicators)) * 15)

        # Calculate Complete Matrix
        total_score = geo_score + capability_score + industry_score + scale_score

        should_generate_proposal = total_score >= 75
